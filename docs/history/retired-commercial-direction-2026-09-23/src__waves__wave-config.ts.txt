/**
 * Wave Configuration — the heartbeat of the marketing system.
 *
 * Each wave is a campaign phase triggered by install milestones.
 * Content rotates through post templates to avoid repetition.
 *
 * WAVE 0 — Bootstrap (< 15 installs):
 *   Pure awareness. "We built something. Here's what it does."
 *   Focus: discovery by Shopify developers and early adopters.
 *
 * WAVE 1 — Growth (15–49 installs, ≥ 2 installs/week):
 *   Social proof emerges. "X merchants already using it."
 *   Focus: features, use cases, early adopter stories.
 *
 * WAVE 2 — Momentum (50–99 installs, ≥ 5 installs/week):
 *   Community and reviews. "Here's what merchants say."
 *   Focus: testimonials, review solicitation, community posts.
 *
 * WAVE 3 — Scale (100+ installs, ≥ 10 installs/week):
 *   Conversion and retention. Direct offer posts, tier upgrades.
 *   Focus: CTAs, the 3-month Pro offer, merchant milestones.
 *
 * RETREAT: if velocity drops below threshold for 14+ days → step back a wave.
 */

export interface WaveConfig {
  id: number;
  name: string;
  description: string;
  installThreshold: number;       // Total installs needed to enter this wave
  velocityThreshold: number;      // Weekly install velocity to maintain the wave
  retreatAfterDays: number;       // Days of low velocity before retreating
  postingSchedule: PostingSchedule;
  channels: ChannelPriority;
  contentTags: string[];          // Which content tags to pull from the library
}

export interface PostingSchedule {
  postsPerWeek: number;
  bestDays: number[];             // 0=Sun, 1=Mon … 6=Sat
  bestHours: number[];            // UTC hours
}

export interface ChannelPriority {
  primary: string[];              // Post to all of these on every scheduled post
  secondary: string[];            // Rotate through these
}
